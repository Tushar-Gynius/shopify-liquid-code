	{%- assign pincodeList = settings.pincode_list | strip_newlines | strip | split: ',' -%} 

	document.addEventListener("DOMContentLoaded", function() {
	  var pincodePopup = document.getElementById('pincodePopup');
	  var pincodeOverlay = document.createElement('div');
	  pincodeOverlay.classList.add('pincodeOverlay');
	  var closePopup = document.getElementById('closePopup');
	  var pincodeInput = document.getElementById('pincodeInput');
	  var availabilityMsg = document.querySelector(".availabilityMsg");
	  var pincodeValue = document.querySelectorAll('.pincodeValue');
	  var changePincode = document.querySelectorAll('.changePincode');
	  
	  window.showPopup = function showPopup() {
		pincodePopup.style.display = 'block';
		document.body.appendChild(pincodeOverlay);
		pincodeOverlay.style.display = 'block';
		document.body.style.overflow = 'hidden';
	  }
	  
	  function hidePopup() {
		pincodePopup.style.display = 'none';
		pincodeOverlay.style.display = 'none';
		document.body.removeChild(pincodeOverlay);
		document.body.style.overflow = 'auto';
	  }
	  
	  function getPincode() {
		return localStorage.getItem('Pincode') || '';
	  }

	  function setPincode(pincode, available) {
		if(available) {
		  localStorage.setItem('Pincode', pincode);
		}else {
		  localStorage.removeItem('Pincode');
		}
	  }

	  pincodeInput.value = getPincode();

	  pincodeValue.forEach(function(element) {
		if(pincodeInput.value) {
		  element.textContent = pincodeInput.value;
		}else {
		  element.textContent = 'Pincode';
		}
	  });

	  pincodeOverlay.addEventListener('click', hidePopup);    
	  closePopup.addEventListener('click', hidePopup);

	  var pincodeList = {{ pincodeList | json }};
	  
	  pincodeInput.addEventListener('input', function(event) {
		var inputValue = event.target.value;
		var removeExtra = inputValue.replace(/\D/g, '');
		var maxLength = 6; 
		
		removeExtra = removeExtra.substring(0, maxLength);
		event.target.value = removeExtra;

		if(removeExtra.length < maxLength) {
		  availabilityMsg.textContent = "Enter a Valid Pincode";
		  availabilityMsg.style.color = "red";
		}else {
		  availabilityMsg.textContent = "";
		}

		if(pincodeList.includes(removeExtra)) {
		  setPincode(removeExtra, true);
		}else {
		  setPincode(removeExtra, false);
		}
	  });

	  document.getElementById("checkAvailability").addEventListener("click", function() {
		var pincode = pincodeInput.value;
		
		if(!pincode.match(/^\d{6}$/)) {
		  availabilityMsg.textContent = "Enter a Valid Pincode";
		  availabilityMsg.style.color = "red";
		  return;
		}
		
		var isAvailable = pincodeList.includes(pincode);

		if(isAvailable) {
		  availabilityMsg.textContent = "Delivery is Available";
		  availabilityMsg.style.color = "green";
		}else {
		  availabilityMsg.textContent = "Delivery is Not Available";
		  availabilityMsg.style.color = "red";
		}

		pincodeValue.forEach(function(element) {
		  if(pincode) {
			element.textContent = pincode;
		  }else {
			element.textContent = 'Pincode';
		  }
		});

		setPincode(pincode, isAvailable);
	  });

	  changePincode.forEach(function(button) {
		button.addEventListener('click', function() {
		  showPopup();
		});
	  });
	});
