	<script>
	  document.addEventListener("DOMContentLoaded", function() {
		var endlessScroll = new Ajaxinate({
		  container: '#product-list-{{ section.id }}',
		  pagination: '#AjaxinatePagination',
		  loadingText: 'Loading...',
		  method: 'click'
		});
	  });
	  
	  var totalPro = parseInt(`{{ collection.products_count }}`); // This is By Default Object of Shopify
	  var currPro = parseInt(`{{ section.settings.products_per_page }}`); // Create Settings ("type": "text") of No. of Products For Collection Page
	  var countPro = parseInt(`{{ section.settings.products_per_page }}`); // Create Settings ("type": "text") of No. of Products For Collection Page
	  
	  $('body').on('click', '#load-more', function (e) {
		e.preventDefault();
		const currPage = $(this).attr('page');
		const lastPage = currPage * countPro;
		
		if(totalPro < lastPage) {
		  let remainCount = lastPage - totalPro;
		  remainCount = countPro - remainCount;
		  currPro = currPro + remainCount;
		}else {
		  currPro = currPro + countPro;
		}
		$('#current-count').text(currPro);
	  })
	</script>
