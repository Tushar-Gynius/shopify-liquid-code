	$(document).ready(function() {
	  // Cart Drawer Discounts
	  function applyButton() {
		var url = window.location.origin;
		var discountValue = $(this).attr('discount');
		var self = this;

		fetch(`${url}/discount/${discountValue}`, {method: 'GET'})
		.then(res => res.text())
		.then(response => {
		  updateCartDrawer();
		  localStorage.setItem('Discount', discountValue);
		})
		.catch(error => {
		  console.error('Error:', error);
		});
	  }

	  function discountText() {
		$.getJSON('/cart', function(cartItem) {
		  var cartTotal = cartItem.total_price / 100;
		  var appliedDiscount = localStorage.getItem('Discount');
			
		  if(appliedDiscount === '10Percent' && cartTotal <= 499) {
			localStorage.removeItem('Discount');
			$('.applied').each(function() {
			  if($(this).attr('discount') === '10Percent') {
				$(this).text('Apply');
			  }
			});
		  }else if(appliedDiscount === '20Percent' && cartTotal <= 799) {
			localStorage.removeItem('Discount');
			$('.applied').each(function() {
			if($(this).attr('discount') === '20Percent') {
			  $(this).text('Apply');
			}
			});
		  }else {
			$('.applied').each(function() {
			  if ($(this).attr('discount') === appliedDiscount) {
				$(this).text('Applied');
			  }
			});
		  }
		});
	  }
  
	  // Cart Drawer Free Product Above 399
	  function applyGwpOffer(cartItem) {
		var totalcartprice = parseInt(cartItem.total_price / 100);
		var totalUpdatedPrice = totalcartprice >= 399 ? totalcartprice : totalcartprice + 49; // Cart Drawer Shipping Charge
		var gwp_product_var_id = 40112532815917;
		var cart_item = JSON.stringify(cartItem.items);
		let cart_item_match = cart_item.match(gwp_product_var_id);

		if(totalUpdatedPrice >= 399) {
		  if(cart_item_match == null) {
			add_free_item(gwp_product_var_id, 1);
		  }
		}else {
		  if(cart_item_match != null) {
			remove_free_item(gwp_product_var_id);
		  }
		}
	  }

	  // Cart Drawer Add Free Product
	  function add_free_item(id, quantity) {
		$.ajax({
		  type: 'POST',
		  url: '/cart/add.js',
		  data: { id: id, quantity: quantity },
		  dataType: 'json',
		  success: function(cart) {
			updateCartDrawer();
		  },
		  error: function(err) {
			console.warn(err);
		  }
		});
	  }

	  // Cart Drawer Remove Free Product
	  function remove_free_item(id) {
		$.ajax({
		  type: 'POST',
		  url: '/cart/change.js',
		  data: { id: id, quantity: 0 },
		  dataType: 'json',
		  success: function(cart) {
			updateCartDrawer();
		  },
		  error: function(err) {
			console.log('err', err);
		  }
		});
	  }
	  
	  // Cart Drawer Open
	  function openCartDrawer() {
		$(".cart-drawer").addClass("cart-drawer--active");
	  }

	  // Cart Drawer Close
	  function closeCartDrawer() {
		$(".cart-drawer").removeClass("cart-drawer--active");
	  }

	  // Cart Drawer Item Count
	  function updateCartItemCounts(count) {
		$(".cart-count").each(function() {
		  $(this).text(count);
		});
	  }

	  // Cart Drawer Update
	  async function updateCartDrawer() {
		const res = await fetch("/?section_id=drawer"); // Sections Name
		const text = await res.text();
		const html = $(document.createElement("div"));
		html.html(text);
		const newBox = html.find(".cart-drawer").html();
		
		$(".cart-drawer").html(newBox);
		$('.applied').off('click').on('click', applyButton); // Cart Drawer Discounts
		discountText(); // Cart Drawer Discounts
		addCartDrawerListeners();
	  }

	  function addCartDrawerListeners() {
		// Cart Drawer Quantity Button
		$(".cart-drawer-quantity-selector button").on("click", async function() {
		  const rootItem = $(this).closest(".cart-drawer-item");
		  const key = rootItem.data("line-item-key");
		  const currentQuantity = Number($(this).parent().find("input").val());
		  const isUp = $(this).hasClass("cart-drawer-quantity-selector-plus");
		  const newQuantity = isUp ? currentQuantity + 1 : currentQuantity - 1;

		  const res = await fetch("/cart/update.js", {
			method: "post",
			headers: {
			  Accept: "application/json",
			  "Content-Type": "application/json",
			},
			body: JSON.stringify({ updates: { [key]: newQuantity } }),
		  });
		  
		  const cart = await res.json();
		  updateCartItemCounts(cart.item_count);
		  updateCartDrawer();
		  applyGwpOffer(cart);
		});

		// Cart Drawer Remove Button
		$(".cart-drawer-item-remove").on("click", async function() {
		  const rootItem = $(this).closest(".cart-drawer-item");
		  const key = rootItem.data("line-item-key");
	  
		  const res = await fetch("/cart/change.js", {
			method: "post",
			headers: {
			  Accept: "application/json",
			  "Content-Type": "application/json",
			},
			body: JSON.stringify({ id: key, quantity: 0 }),
		  });
	  
		  const cart = await res.json();
		  updateCartItemCounts(cart.item_count);
		  updateCartDrawer();
		  applyGwpOffer(cart);
		});

		// Cart Drawer Prevent Click
		$(".cart-drawer-box").on("click", function(e) {
		  e.stopPropagation();
		});

		// Cart Drawer Close Outside Click
		$(".cart-drawer-header-right-close, .cart-drawer").on("click", function() {
		  closeCartDrawer();
		});
	  }

	  // Cart Drawer ATC
	  $('form[action="/cart/add"]').on("submit", async function(e) {
		e.preventDefault();

		await fetch("/cart/add", {
		  method: "post",
		  body: new FormData(this),
		});

		const res = await fetch("/cart.js");
		const cart = await res.json();
		updateCartItemCounts(cart.item_count);
		await updateCartDrawer();
		openCartDrawer();
		applyGwpOffer(cart);
	  });

	  // Cart Drawer Bag Icon
	  $('a[href="/cart"]').on("click", function(e) {
		e.preventDefault();
		openCartDrawer();
	  });

	  // Cart Drawer Initial Setup
	  addCartDrawerListeners();

	  // Cart Drawer Free Product
	  $.getJSON('/cart', function(cartItem) {
		applyGwpOffer(cartItem);
	  });
	  
	  // Cart Drawer Discounts
	  $('.applied').on('click', applyButton);
	  discountText();
	});
